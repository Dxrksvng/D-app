Flow ของ Pilot

1Connectwallet.useProvider() (MetaMask Connect)
2วางหรือสแกน QRparsePaymentUri()
3วิเคราะห์GET /v1/risk 0.4–6 วินาที
4Popupdecide(risk) = UI ของ Frontend
5MetaMask Reviewส่งพร้อมตราใน data
6Verified/v1/verify/tx · Trust Score
import { RiskGuardClient, WalletManager, guardedSend, parsePaymentUri, units } from '/sdk/riskguard.js';

const api = new RiskGuardClient({ apiBase: 'https://api.bridgesecure.com' });
const wallet = new WalletManager({ chains: await api.getChains() });
await wallet.useProvider(metamaskConnectProvider, { name: 'MetaMask' });   // มือถือ
// desktop: await wallet.discover(); await wallet.connect(uuid);

const p = parsePaymentUri(qrText);   // 0x… หรือ ethereum:0x…@43113?value=…
const r = await guardedSend({
  api, wallet, chainId: p.chainId ?? 43113, to: p.address, valueWei: units.parse(amount),
  decide: (risk) => showBridgeAlert(risk),   // UI ของ Frontend: true = โอนต่อ, false = ยกเลิก
  onStage: (stage, info) => updateProgress(stage, info),
});
// r.status 'cancelled' | 'confirmed' | 'failed' · r.risk · r.txHash · r.verification

ข้อความบน popup ต้องไม่เกินหลักฐาน เช่น ห้ามเขียน "ฟอกเงิน" ถ้า signal ไม่ได้ระบุแบบนั้น · เราเตือน ไม่บล็อก ผู้ใช้ต้องโอนต่อได้เสมอ

ตรา "Verified by BridgeSecure" บนทุกธุรกรรม

  • ที่เก็บ: อยู่ใน input data ของธุรกรรมโอนเอง (86 bytes, gas เพิ่มประมาณ 1,400) เปิด Explorer → Input Data → UTF-8 จะเห็น BRIDGESECURE|v1|c=43113|l=LOW|ts=95|…|k=…
  • ปลอมหรือคัดลอกไม่ได้: ลายเซ็น HMAC ผูกกับ chain, ผู้ส่ง, ผู้รับ และจำนวนเงิน
  • หน้ายืนยัน: verify.html?chain_id=43113&tx=0x… หรือใส่ที่อยู่กระเป๋าเพื่อดูทุกธุรกรรมที่มีตรา
  • Backend ไม่เก็บข้อมูลธุรกรรม: อ่านจาก chain ใหม่ทุกครั้ง · ใส่ตราเฉพาะเมื่อปลายทางเป็นกระเป๋า (ไม่ใช่ contract) · ปิดได้ด้วย attest: false
  • ใช้คำว่า "Verified by BridgeSecure" ไม่ใช่ "Validator" เพราะเราไม่ได้เป็น validator ของ blockchain
